      try{
        $cal=Get-Content -Raw -LiteralPath $calPath|ConvertFrom-Json
        if($cal.blocking_acceptance-eq$true){$pass=$true;break}
        elseif($cal.blocking_acceptance-eq$false){throw ('GPU calibration explicitly rejected: '+($cal|ConvertTo-Json -Compress -Depth 20))}
      }catch{if($_.Exception.Message-like'GPU calibration explicitly rejected*'){throw}}
    }
    Start-Sleep -Seconds 2
  }
  if(-not$pass){throw 'GPU R065 RC16 did not produce calibration PASS within 7 minutes'}
  Start-Sleep -Seconds 3
  if($sup.HasExited){throw 'GPU supervisor exited immediately after calibration PASS'}
  if(-not(Test-Path -LiteralPath $health)){throw 'GPU health missing after calibration PASS'}
  $gh=Get-Content -Raw -LiteralPath $health|ConvertFrom-Json
  if([string]$gh.mode-ne'ACTIVE_CONTINUOUS'){throw ('GPU mode after calibration is '+[string]$gh.mode+', expected ACTIVE_CONTINUOUS')}
  if([int]$gh.loop_samples-lt5){throw 'GPU active loop sample canary not satisfied'}
  $activation.bootstrap_result=if(Test-Path -LiteralPath $bootPath){Get-Content -Raw -LiteralPath $bootPath|ConvertFrom-Json}else{$null}
  $activation.warmup_result=if(Test-Path -LiteralPath $warmPath){Get-Content -Raw -LiteralPath $warmPath|ConvertFrom-Json}else{$null}
  $activation.calibration_result=$cal
  $activation.gpu_health=$gh
  $activation.gpu_active=$true
  $activation.checks.gpu_runtime='PASS_CALIBRATED_LIVE'
  Pass-MainLearningStage 'GPU_RUNTIME' 'Start one supervisor that bootstraps, measures 40 warmup plus 10/25/40 phases, then promotes live only on real evidence' 'GPU_LIVE_PASS' 'GPU supervisor remained alive after full calibrated PASS and reported ACTIVE_CONTINUOUS.' @{
    supervisor_pid=[int]$sup.Id
    mode=[string]$gh.mode
    loop_samples=[int]$gh.loop_samples
    calibration_pass=[bool]$cal.blocking_acceptance
    minimum_stable_productive_duty=$cal.minimum_stable_productive_duty
    measured_control_effect=[bool]$cal.measured_control_effect
  }

  $currentMainLearningStage='CPU_RC6_COEXISTENCE_POST'
  Start-MainLearningStage 'CPU_RC6_COEXISTENCE_POST' 'Prove GPU activation did not stop or stall CPU RC6'
  $cpuPost=Assert-CpuRc6LoopLiveness $cpuState $cpuHealth $cpuRel 9
  if([int]$cpuPost.loop_after-le[int]$activation.cpu_rc6_preflight.loop_after){throw 'CPU RC6 did not advance beyond pre-GPU coexistence proof'}
  $activation.cpu_rc6_post_gpu=$cpuPost
  $activation.checks.cpu_rc6_post_gpu='PASS_ADVANCING'
  Pass-MainLearningStage 'CPU_RC6_COEXISTENCE_POST' 'Prove GPU activation did not stop or stall CPU RC6' 'CPU_RC6_POST_PASS' 'CPU RC6 remained ACTIVE_CONTINUOUS and advanced after GPU live promotion.' ([hashtable]$cpuPost)

  $activation.cpu_rc6_untouched=$true
  Save-Activation 'PASS_GPU_LIVE_LEARNING_ACTIVE'
  Complete-MiningSniperLearningRun -ProjectRoot $Root -RunId $learningRunId -Release $Release -Outcome PASS -ReasonCode 'GPU_LIVE_AND_LEARNING_ACTIVE' -ReasonText 'GPU R065 RC16 completed exact duty fixture, calibrated 10/25/40 and entered live mode while CPU RC6 remained independent.' -Evidence @{
    activation_report=$activationPath
    gpu_active=$true
    minimum_stable_productive_duty=$activation.calibration_result.minimum_stable_productive_duty
    cpu_rc6_untouched=$true
  } -ComponentOutcomes @{learning_store='PASS';cpu_rc6='PASS_UNTOUCHED';gpu_live='PASS';duty_control_surface='PASS'}|Out-Null
  Write-Host '';Write-Host 'MININGSNIPER R065 GPU RC16: PASS / GPU LIVE + LEARNING ACTIVE' -ForegroundColor Green
  Write-Host ('GPU MINIMUM STABLE PRODUCTIVE DUTY: '+$activation.calibration_result.minimum_stable_productive_duty)
  Write-Host ('ACTIVATION REPORT: '+$activationPath)
  Write-Host ('LEARNING BASE: '+$learningBasePath)
  Write-Host ('LEARNING JOURNAL: '+$learningJournalPath)
  exit 0
}catch{
  $errObj=$_
  $msg=$errObj.Exception.Message
  $activation.error=[ordered]@{stage=$currentMainLearningStage;message=$msg;exception=$errObj.Exception.ToString();script_stack=$errObj.ScriptStackTrace;position=$errObj.InvocationInfo.PositionMessage}
  $activation.cpu_rc6_untouched=$true
  $activation.gpu_active=$false
  try{Save-Activation 'FAIL_CLOSED'}catch{}
  Fail-MainLearningStage ([string]$currentMainLearningStage) $msg @{exception_type=$errObj.Exception.GetType().FullName;exception_message=$msg;script_stack=$errObj.ScriptStackTrace;position_message=$errObj.InvocationInfo.PositionMessage;activation_report=$activationPath;gpu_started=[bool]$activation.gpu_started;gpu_active=$false}
  try{
    if($learningReady){
      Complete-MiningSniperLearningRun -ProjectRoot $Root -RunId $learningRunId -Release $Release -Outcome FAIL -ReasonCode 'ACTIVATION_EXCEPTION' -ReasonText $msg -Evidence @{exception_type=$errObj.Exception.GetType().FullName;exception_message=$msg;script_stack=$errObj.ScriptStackTrace;position_message=$errObj.InvocationInfo.PositionMessage;activation_report=$activationPath;gpu_started=[bool]$activation.gpu_started;gpu_active=$false;current_stage=$currentMainLearningStage} -ComponentOutcomes @{learning_store='PASS_UNLESS_FAILURE_OCCURRED_INSIDE_LEARNING_BOOTSTRAP';cpu_rc6_untouched=$true;gpu_started=[bool]$activation.gpu_started;gpu_live=$false}|Out-Null
    }else{
      Write-BootstrapLearningEvent 'RUN_FINAL_FAIL' 'RUN' 'FAIL' 'Finalize runtime attempt' 'ACTIVATION_EXCEPTION' $msg @{activation_report=$activationPath;gpu_started=[bool]$activation.gpu_started;gpu_active=$false;current_stage=$currentMainLearningStage;exception_type=$errObj.Exception.GetType().FullName}|Out-Null
    }
  }catch{}
  if($activation.gpu_started){
    try{Set-Content -LiteralPath (Join-Path $stateDir 'STOP_REQUESTED_GPU_R065_RC16') -Value ((Get-Date).ToUniversalTime().ToString('o')) -Encoding ascii}catch{}
  }
  Write-Host '';Write-Host ('MININGSNIPER R065 GPU RC16: FAIL - '+$msg) -ForegroundColor Red
  Write-Host 'CPU R063 RC6 WAS NOT STOPPED OR RECONFIGURED.'
  Write-Host ('ACTIVATION REPORT: '+$activationPath)
  Write-Host ('LEARNING BASE: '+$learningBasePath)
  Write-Host ('LEARNING JOURNAL: '+$learningJournalPath)
  Write-Host ('LEARNING RUN ID: '+$learningRunId)
  Write-Host 'Do not rerun after FAIL; return activation report + learning base.'
  exit 1
}
