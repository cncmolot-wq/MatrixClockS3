  elseif($clamped-gt1.0){$clamped=1.0}

    cycle_ms=$CycleLengthMs
    active_ms=$activeMs
  }
}

function Convert-LolApiHashrateMhs([object]$Api){
  if($null-eq$Api -or -not$Api.ok -or $null-eq$Api.data){return $null}
  try{
    foreach($alg in @($Api.data.Algorithms)){
      if($null-eq$alg){continue}
      $name=[string]$alg.Algorithm
      if($name-notmatch'(?i)^Etchash$'){continue}
      $perfProp=$alg.PSObject.Properties['Total_Performance']
      $unitProp=$alg.PSObject.Properties['Performance_Unit']
      if($null-eq$perfProp -or $null-eq$unitProp){continue}
      $v=[double]$perfProp.Value
      if($v-le0){continue}
      $unit=([string]$unitProp.Value).Trim().ToLowerInvariant()
      switch -Regex($unit){
        '^mh/s$' {return $v}
        '^mhs$'  {return $v}
        '^kh/s$' {return ($v/1000.0)}
        '^h/s$'  {return ($v/1000000.0)}
        '^gh/s$' {return ($v*1000.0)}
      }
    }
  }catch{}
  return $null
}
function Parse-LolHashrateLine([string]$Line){

  # Exact lolMiner 1.98a fixture: "Average speed (5s): 17.89 Mh/s"
  $m=[regex]::Match($Line,'(?i)\bAverage\s+speed(?:\s*\([^)]*\))?\s*:\s*([0-9]+(?:[\.,][0-9]+)?)\s*(Mh/s|Mhs)\b')
  if($m.Success){
    return [double]::Parse($m.Groups[1].Value.Replace(',','.'),[Globalization.CultureInfo]::InvariantCulture)
  }
  return $null
}
