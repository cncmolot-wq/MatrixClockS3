-Force -ErrorAction SilentlyContinue}}catch{}
    $script:samplerProc=$null
  }
}
function Parse-SamplerRowsSince([int]$StartLine){
  $rows=[Collections.Generic.List[object]]::new()
  foreach($line in @(Get-LinesSince $script:samplerFile $StartLine)){
    try{
      $a=([string]$line)-split ','|ForEach-Object{$_.Trim()}
      if($a.Count-lt8){continue}
      $u=Parse-Num $a[1];$mu=Parse-Num $a[2];$used=Parse-Num $a[3];$total=Parse-Num $a[4];$temp=Parse-Num $a[5];$power=Parse-Num $a[6]
      if($null-eq$u -or $null-eq$power){continue}
      $rows.Add([pscustomobject][ordered]@{
        timestamp=$a[0];util_pct=$u;mem_util_pct=$mu;memory_used_mb=$used;memory_total_mb=$total;
        temp_c=$temp;power_w=$power;pstate=$a[7]
      })
    }catch{}
  }
  return @($rows)
}
function Get-ApiEtchashSnapshot{
  $api=Get-LolApi
  if($null-eq$api-or-not$api.ok){return [pscustomobject][ordered]@{ok=$false;performance_mhs=$null;accepted=$null;rejected=$null;stales=$null}}
  try{
    foreach($alg in @($api.data.Algorithms)){
      if([string]$alg.Algorithm -match '(?i)^Etchash$'){
        return [pscustomobject][ordered]@{
          ok=$true
          performance_mhs=(Convert-LolApiHashrateMhs $api)
          accepted=[int64]$alg.Total_Accepted
          rejected=[int64]$alg.Total_Rejected
          stales=[int64]$alg.Total_Stales
        }
      }
    }
  }catch{}
  return [pscustomobject][ordered]@{ok=$false;performance_mhs=$null;accepted=$null;rejected=$null;stales=$null}
}
function Measure-DutyPhase([double]$DutyFraction,[string]$Name,[int]$SettleCycles=4,[int]$MeasureCycles=10){
  for($i=0;$i-lt$SettleCycles;$i++){
    if($miner.HasExited){throw ('lolMiner exited during '+$Name+' settle')}
    $null=Run-DutyCycle $DutyFraction ($Name+'_SETTLE') 2000 $false
  }

  $samplerStart=Get-LineCount $script:samplerFile
  $logStart=Get-LineCount ([string]$r.native_log_file)
  $apiStart=Get-ApiEtchashSnapshot
  $samples=[Collections.Generic.List[object]]::new()

  for($i=0;$i-lt$MeasureCycles;$i++){
    if($miner.HasExited){throw ('lolMiner exited during '+$Name+' measure')}
    $samples.Add((Run-DutyCycle $DutyFraction $Name 2000 $false))
  }
  Start-Sleep -Milliseconds 500

  $samplerRows=@(Parse-SamplerRowsSince $samplerStart)
  $freshHash=@(Get-FreshLogHashratesSince $logStart)
  $apiEnd=Get-ApiEtchashSnapshot

  $power=@($samplerRows|Where-Object{$null-ne$_.power_w}|ForEach-Object{[double]$_.power_w})
  $util=@($samplerRows|Where-Object{$null-ne$_.util_pct}|ForEach-Object{[double]$_.util_pct})
  $temp=@($samplerRows|Where-Object{$null-ne$_.temp_c}|ForEach-Object{[double]$_.temp_c})
  $awake=@($samples|ForEach-Object{[double]$_.actual_duty_fraction})
  $poolTrue=@($samples|Where-Object{Test-Pool20128 $_.network}).Count
  $threadMin=($samples|Measure-Object -Property suspended_threads -Minimum).Minimum
  $hashPositive=@($freshHash|Where-Object{$_-gt0})

  $freshMedian=Median ([double[]]$hashPositive)
  $freshMean=Average ([double[]]$hashPositive)
  $poolFraction=if($samples.Count-gt0){$poolTrue/[double]$samples.Count}else{0}
  $actualDutyMean=Average ([double[]]$awake)
  $actualDutyError=if($null-ne$actualDutyMean){[Math]::Abs($actualDutyMean-$DutyFraction)}else{1.0}

  return [pscustomobject][ordered]@{
    name=$Name
    duty=$DutyFraction
    settle_cycles=$SettleCycles
    measure_cycles=$MeasureCycles
    sampler_samples=$samplerRows.Count
    actual_duty_mean=$actualDutyMean
    actual_duty_error=$actualDutyError
    sampler_util_mean_pct=(Average ([double[]]$util))
    sampler_util_median_pct=(Median ([double[]]$util))
    sampler_power_mean_w=(Average ([double[]]$power))
    sampler_power_median_w=(Median ([double[]]$power))
    sampler_temp_max_c=if($temp.Count-gt0){($temp|Measure-Object -Maximum).Maximum}else{$null}
    fresh_hash_samples=$freshHash.Count
    fresh_positive_hash_samples=$hashPositive.Count
    fresh_hash_median_mhs=$freshMedian
    fresh_hash_mean_mhs=$freshMean
    pool_ready_fraction=$poolFraction
    min_suspended_threads=[int]$threadMin
    api_start=$apiStart
    api_end=$apiEnd
  }
}
